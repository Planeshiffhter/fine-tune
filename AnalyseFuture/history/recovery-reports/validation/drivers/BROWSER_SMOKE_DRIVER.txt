from pathlib import Path
import subprocess,os,re,json
from playwright.sync_api import sync_playwright, expect
root=Path('/workspace/fine-tune')
server=subprocess.Popen(['node','apps/api/src/server.mjs'],cwd=root,env={**os.environ,'HOST':'127.0.0.1','PORT':'0'},stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True)
try:
 line=server.stdout.readline(); m=re.search(r'http://127\.0\.0\.1:\d+',line);assert m,line
 base=m.group(0)
 with sync_playwright() as p:
  browser=p.chromium.launch(executable_path='/usr/bin/chromium',headless=True)
  page=browser.new_page(accept_downloads=True,viewport={'width':1440,'height':1000})
  errors=[];page.on('pageerror',lambda e:errors.append(str(e)))
  health=page.request.get(base+'/api/health');assert health.status==200
  assert health.json()=={'status':'ok','mode':'offline_mock','model_connected':False}
  page.goto(base,wait_until='networkidle');expect(page).to_have_title('fine-tune · Builder prototype')
  expect(page.locator('#download')).to_be_disabled();assert page.locator('#preview').get_attribute('sandbox')==''
  for label,category in [(None,'portfolio'),('Boutique store','store'),('Analytics dashboard','dashboard'),('Neighborhood cafe','restaurant')]:
   if label:page.get_by_role('button',name=label).click()
   text=page.locator('#prompt').input_value();expect(page.locator('#prompt-count')).to_have_text(f'{len(text)} / 800')
   with page.expect_response(lambda r:r.url.endswith('/api/generate') and r.request.method=='POST') as received:page.locator('#generate').click()
   response=received.value;assert response.status==200;data=response.json();assert data['category']==category;assert data['status']=='mock_only'
   expect(page.locator('#status')).to_have_text('Mock preview created. No code was executed or saved.')
   expect(page.locator('#template-label')).to_contain_text(data['title'])
   expect(page.frame_locator('#preview').locator('h1')).to_be_visible()
   assert page.locator('#plan .plan-step').count()==len(data['plan'])
   expect(page.locator('#download')).to_be_enabled()
  with page.expect_download() as saved:page.locator('#download').click()
  download=saved.value;assert download.suggested_filename=='index.html';content=Path(download.path()).read_text();assert '<!doctype html>' in content.lower();assert content==data['files'][0]['content']
  def fail(route):route.fulfill(status=400,content_type='application/json',body=json.dumps({'error':'Controlled browser error fixture'}))
  page.route('**/api/generate',fail);page.locator('#generate').click();expect(page.locator('#status')).to_have_text('Controlled browser error fixture')
  expect(page.locator('#download')).to_be_disabled();assert page.locator('#preview').get_attribute('srcdoc') is None
  expect(page.locator('#plan')).to_have_text('No verified output is available.')
  page.unroute('**/api/generate',fail);page.locator('#generate').click();expect(page.locator('#download')).to_be_enabled()
  page.set_viewport_size({'width':390,'height':844});expect(page.locator('#prompt')).to_be_visible();expect(page.locator('#generate')).to_be_visible()
  page.screenshot(path='/tmp/fine-tune-ui-smoke.png',full_page=True)
  assert not errors,errors
  browser.close()
 print('PASS: headless Chromium UI; initial state; four real template flows; sandboxed rendered frame; plan; character count; HTML download; controlled error and recovery; mobile controls; zero uncaught page errors')
finally:
 server.terminate()
 try:server.wait(timeout=5)
 except subprocess.TimeoutExpired:server.kill();server.wait()
